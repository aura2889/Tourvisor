<script>
    {
        const qr = document.querySelector('.trv-mobile-app-banner__qr');

        qr.addEventListener('click', () => {
            const modal = trvCreateModalBlock();
            const modalBlock = document.querySelector('.trv-mobile-app-banner-modal');
            const modalBlockClone = modalBlock.cloneNode(true);

            modalBlockClone.classList.add('trv-mobile-app-banner-modal--active');
            modal.appendChild(modalBlockClone);

            trvOpenModalBlock({ closeIcon: false });

            const closeButton = modal.querySelector('.trv-mobile-app-banner-modal-close');
            if (closeButton) {
                closeButton.addEventListener('click', (e) =>{
                    bdScroll.enable();
                    modal.classList.remove('trv-modal--active');
                    setTimeout(() => {
                        modal.remove()
                    }, 300)
                })
            }

            const modalButtonCopyLink = modal.querySelector(
                '.trv-mobile-app-banner-modal-button',
            );
            const originalText = modalButtonCopyLink.textContent;
            let isCopying = false;

            modalButtonCopyLink.addEventListener('click', async (event) => {
                event.preventDefault();

                if (isCopying) {
                    return;
                }

                isCopying = true;
                modalButtonCopyLink.disabled = true;

                try {
                    await navigator.clipboard.writeText('https://onelink.to/zk8s3s');
                    modalButtonCopyLink.textContent = 'Скопировано';

                    setTimeout(() => {
                        isCopying = false;
                        modalButtonCopyLink.disabled = false;
                        modalButtonCopyLink.textContent = originalText;
                    }, 5000);
                } catch (error) {
                    isCopying = false;
                    modalButtonCopyLink.disabled = false;

                    console.error('Не удалось скопировать ссылку:', error);
                }
            });
        });
    }
</script>
