<script>
    (function () {
        "use strict";

        const SELECTORS = {
            dropdown: ".trv-country-choose-dropdown",
            overlay: ".trv-overlay",
            desc: ".trv-country-choose-dropdown__desc",
            applyBtn: ".trv-country-choose-dropdown__apply",
            closeBtn: ".trv-country-choose-dropdown__close",
            headerLang: ".trv-header__language",
            headerLangMobile: ".trv-header__language-mobile",
        };

        const CSS = {
            dropdownActive: "trv-country-choose-dropdown--active",
            overlayActive: "trv-overlay--active",
            headerLangActive: "trv-header__language--country-choose",
            headerMobileActive: "trv-header__language-mobile--country-choose",
        };

        const ALLOWED_COUNTRY_CODES = ["by", "kz", "uz", "kg"];
        const COUNTRY_CODE_MAP = { kz: "kz", by: "by", uz: "uz", kg: "kg" };

        const POPOVER_OFFSET_TOP = 13;
        const POPOVER_OFFSET_LEFT = 226;
        const POPOVER_SCREEN_MARGIN = 8;
        const RESIZE_DEBOUNCE_MS = 100;

        function getElements() {
            return {
                dropdown: document.querySelector(SELECTORS.dropdown),
                overlay: document.querySelector(SELECTORS.overlay),
                desc: document.querySelector(SELECTORS.desc),
                applyBtn: document.querySelector(SELECTORS.applyBtn),
                closeBtn: document.querySelector(SELECTORS.closeBtn),
                headerLang: document.querySelector(SELECTORS.headerLang),
                headerLangMobile: document.querySelector(SELECTORS.headerLangMobile),
            };
        }

        async function fetchGeoIp() {
            const res = await fetch("https://tourvisor.ru/country/geoip");
            if (!res.ok) {
                return null;
            }

            return res.json();
        }

        async function resolveIpCountry() {
            const data = await fetchGeoIp();
            if (!data || !data.country_code) {
                return null;
            }

            const countryEnum = COUNTRY_CODE_MAP[data.country_code.toLowerCase()] ?? "ru";
            if (!ALLOWED_COUNTRY_CODES.includes(countryEnum)) {
                return null;
            }

            const country = (await trvGetCountryByEnum(countryEnum)) ?? null;
            if (!country?.namePrepositional) {
                return null;
            }

            return { country, countryEnum };
        }

        function showPopup(els) {
            const { dropdown, overlay } = els;
            if (!dropdown || !overlay) {
                return;
            }

            overlay.classList.add(CSS.overlayActive);
            dropdown.classList.add(CSS.dropdownActive);

            repositionPopover(els);
            watchResizeWhileOpen(els);

            if (typeof bdScroll !== "undefined") {
                bdScroll?.disable?.();
            }
        }

        function hidePopup(els) {
            const { dropdown, overlay, headerLang, headerLangMobile } = els;
            if (!dropdown || !overlay) {
                return;
            }

            overlay.classList.remove(CSS.overlayActive);
            dropdown.classList.remove(CSS.dropdownActive);
            headerLang?.classList.remove(CSS.headerLangActive);
            headerLangMobile?.classList.remove(CSS.headerMobileActive);

            if (typeof bdScroll !== "undefined") {
                bdScroll?.enable?.();
            }
        }

        function repositionPopover(els) {
            const { dropdown, headerLang, headerLangMobile } = els;
            if (!dropdown) {
                return;
            }

            headerLang?.classList.remove(CSS.headerLangActive);
            headerLangMobile?.classList.remove(CSS.headerMobileActive);

            const isDesktopVisible = headerLang && headerLang.offsetWidth > 0 && headerLang.offsetHeight > 0;

            if (isDesktopVisible) {
                placePopoverNear(headerLang, dropdown, true);
                headerLang.classList.add(CSS.headerLangActive);
            } else if (headerLangMobile) {
                placePopoverNear(headerLangMobile, dropdown, false);
                headerLangMobile.classList.add(CSS.headerMobileActive);
            }
        }

        function placePopoverNear(triggerEl, popoverEl, isDesktop) {
            const rect = triggerEl.getBoundingClientRect();
            const popoverWidth = popoverEl.offsetWidth;

            let left = rect.left - (isDesktop ? POPOVER_OFFSET_LEFT : 0);

            const overflowRight = left + popoverWidth - window.innerWidth;
            if (overflowRight > 0) {
                left -= overflowRight + POPOVER_SCREEN_MARGIN;
            }

            popoverEl.style.position = "fixed";
            popoverEl.style.top = `${rect.bottom + POPOVER_OFFSET_TOP}px`;
            popoverEl.style.left = `${left}px`;

            const arrowLeft = rect.left + rect.width / 2 - left;
            popoverEl.style.setProperty("--arrow-left", `${arrowLeft}px`);
        }

        function watchResizeWhileOpen(els) {
            let resizeTimer;

            const handleResize = () => {
                clearTimeout(resizeTimer);
                resizeTimer = setTimeout(() => repositionPopover(els), RESIZE_DEBOUNCE_MS);
            };

            window.addEventListener("resize", handleResize);

            watchOverlayClose(els.overlay, () => {
                window.removeEventListener("resize", handleResize);
            });
        }

        function bindCloseHandlers(els) {
            let closed = false;

            const close = () => {
                if (closed) {
                    return;
                }
                closed = true;
                handleClose(els);
            };

            els.closeBtn?.addEventListener("click", close);
            watchOverlayClose(els.overlay, close);
        }

        function watchOverlayClose(overlay, callback) {
            if (!overlay) {
                return;
            }

            const observer = new MutationObserver(() => {
                if (!overlay.classList.contains(CSS.overlayActive)) {
                    observer.disconnect();
                    callback();
                }
            });

            observer.observe(overlay, { attributeFilter: ["class"] });
        }

        function renderDesc(els, countryNamePrepositional) {
            if (!els.desc) {
                return;
            }

            els.desc.innerHTML = `Похоже, вы сейчас в ${countryNamePrepositional}.<br>Переключить страну?`;
        }

        async function renderCloseButton(els) {
            const { closeBtn } = els;
            if (!closeBtn || !trvDefaultCountryEnum) {
                return;
            }

            const country = await trvGetCountryByEnum(trvDefaultCountryEnum);
            if (country?.nameAccusative) {
                closeBtn.textContent = `Оставить ${country.nameAccusative}`;
            }
        }

        function bindApplyButton(els, country) {
            els.applyBtn?.addEventListener("click", () => applyCountry(country));
        }

        function applyCountry(country) {
            if (!country) {
                return;
            }

            const { id, cityId, currencyId } = country;
            trvUpdateTvSearchDepartureCookie(cityId);
            trvUpdateTrvCookieCurrencyId(currencyId);
            trvUpdateTrvCookieCountryId(id);
            location.reload();
        }

        async function handleClose(els) {
            hidePopup(els);

            if (!trvDefaultCountryEnum) {
                return;
            }

            const country = await trvGetCountryByEnum(trvDefaultCountryEnum);
            if (!country?.id) {
                return;
            }

            const { id, cityId, currencyId } = country;
            trvUpdateTvSearchDepartureCookie(cityId);
            trvUpdateTrvCookieCurrencyId(currencyId);
            trvUpdateTrvCookieCountryId(id);
        }

        async function init() {
            const savedCountryId = trvGetTrvCookieCountryId?.();
            const globalVars = window?.trvGlobalVars || [];
            const isB2b = globalVars.find(item => item.name === 'isB2b')?.value;
            if (savedCountryId || Boolean(isB2b)) {
                return;
            }

            let result;
            try {
                result = await resolveIpCountry();
            } catch (err) {
                return;
            }

            if (!result) {
                return;
            }

            const { country } = result;
            const els = getElements();

            showPopup(els);
            renderDesc(els, country.namePrepositional);
            bindApplyButton(els, country);
            bindCloseHandlers(els);
            await renderCloseButton(els);
        }

        init();
    })();

</script>
